CREATE DATABASE quanlydoan
CHARACTER SET utf8mb4
COLLATE utf8mb4_unicode_ci;

USE quanlydoan;

CREATE TABLE SINHVIEN (
    MSV VARCHAR(20) PRIMARY KEY,
    TENSV VARCHAR(100) NOT NULL,
    KHOA VARCHAR(100) NOT NULL,
    DIEMTB DECIMAL(3,2) NULL,
    LOP VARCHAR(50) NOT NULL,
    EMAIL VARCHAR(100) UNIQUE,
    MATKHAU VARCHAR(255) NULL
);

CREATE TABLE DETAI (
    MDT VARCHAR(20) PRIMARY KEY,
    TENDT VARCHAR(255) NOT NULL,
    MOTA TEXT NULL,
    GIANGVIEN_HUONGDAN VARCHAR(100) NOT NULL,
    SO_LUONG_TOI_DA INT DEFAULT 1,
    TRANG_THAI VARCHAR(30) DEFAULT 'MO'
);

CREATE TABLE DANGKY (
    MDT VARCHAR(20),
    MSV VARCHAR(20),
    NGAY_DANG_KY DATETIME NOT NULL,
    TRANG_THAI VARCHAR(30) DEFAULT 'CHO_DUYET',

    PRIMARY KEY (MDT, MSV),

    CONSTRAINT FK_DANGKY_DETAI
        FOREIGN KEY (MDT)
        REFERENCES DETAI(MDT)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT FK_DANGKY_SINHVIEN
        FOREIGN KEY (MSV)
        REFERENCES SINHVIEN(MSV)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);
