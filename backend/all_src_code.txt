import {
  CanActivate,
  ExecutionContext,
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';

import { JwtService } from '@nestjs/jwt';
import type { Request } from 'express';

@Injectable()
export class JwtAuthGuard implements CanActivate {
  constructor(
    private readonly jwtService: JwtService,
  ) {}

  async canActivate(
    context: ExecutionContext,
  ): Promise<boolean> {
    const request =
      context.switchToHttp().getRequest<Request>();

    const authHeader =
      request.headers.authorization;

    if (!authHeader) {
      throw new UnauthorizedException(
        'Chưa đăng nhập',
      );
    }

    const [type, token] =
      authHeader.split(' ');

    if (
      type !== 'Bearer' ||
      !token
    ) {
      throw new UnauthorizedException(
        'Token không hợp lệ',
      );
    }

    try {
      const payload =
        await this.jwtService.verifyAsync(token);

      (request as Request & { user: any }).user =
        payload;

      return true;
    } catch {
      throw new UnauthorizedException(
        'Token không hợp lệ hoặc đã hết hạn',
      );
    }
  }
}

import {
  CanActivate,
  ExecutionContext,
  ForbiddenException,
  Injectable,
} from '@nestjs/common';

import { Reflector } from '@nestjs/core';

import { ROLES_KEY } from '../decorators/roles.decorator.js';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(
    private readonly reflector: Reflector,
  ) {}

  canActivate(
    context: ExecutionContext,
  ): boolean {
    const requiredRoles =
      this.reflector.getAllAndOverride<string[]>(
        ROLES_KEY,
        [
          context.getHandler(),
          context.getClass(),
        ],
      );

    if (!requiredRoles) {
      return true;
    }

    const request =
      context.switchToHttp().getRequest();

    const user = request.user;

    if (!user) {
      throw new ForbiddenException(
        'Không tìm thấy thông tin người dùng',
      );
    }

    if (!requiredRoles.includes(user.role)) {
      throw new ForbiddenException(
        'Bạn không có quyền truy cập chức năng này',
      );
    }

    return true;
  }
}

import {
  BadRequestException,
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';

import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';

import * as bcrypt from 'bcrypt';
import { JwtService } from '@nestjs/jwt';

import { SinhVien } from './entities/sinhvien.entity.js';
import { RegisterDto } from './dto/register.dto.js';
import { LoginDto } from './dto/login.dto.js';

@Injectable()
export class AuthService {
  constructor(
    @InjectRepository(SinhVien)
    private readonly sinhVienRepository: Repository<SinhVien>,

    private readonly jwtService: JwtService,
  ) {}

  // =========================
  // ĐĂNG KÝ
  // =========================

  async register(data: RegisterDto) {
    const msv = data.MSV.trim();
    const tensv = data.TENSV.trim();
    const email = data.EMAIL.trim();

    // =========================
    // KIỂM TRA MSV
    // =========================

    const msvExists =
      await this.sinhVienRepository.findOne({
        where: {
          MSV: msv,
        },
      });

    if (msvExists) {
      throw new BadRequestException(
        'Mã sinh viên đã được sử dụng',
      );
    }

    // =========================
    // KIỂM TRA EMAIL
    // =========================

    const emailExists =
      await this.sinhVienRepository
        .createQueryBuilder('sinhVien')
        .where(
          'LOWER(sinhVien.EMAIL) = LOWER(:email)',
          {
            email,
          },
        )
        .getOne();

    if (emailExists) {
      throw new BadRequestException(
        'Email đã được sử dụng',
      );
    }

    // =========================
    // HASH MẬT KHẨU
    // =========================

    const hashedPassword =
      await bcrypt.hash(
        data.MATKHAU,
        10,
      );

    // =========================
    // TẠO TÀI KHOẢN
    // =========================

    const sinhVien =
      this.sinhVienRepository.create({
        MSV: msv,

        TENSV: tensv,

        KHOA: null,

        LOP: null,

        DIEMTB: null,

        EMAIL: email,

        MATKHAU: hashedPassword,

        VAITRO: 'SINHVIEN',
      });

    const savedSinhVien =
      await this.sinhVienRepository.save(
        sinhVien,
      );

    // =========================
    // TRẢ KẾT QUẢ ĐĂNG KÝ
    // =========================

    return {
      message:
        'Đăng ký tài khoản thành công',

      user: {
        MSV: savedSinhVien.MSV,

        TENSV: savedSinhVien.TENSV,

        KHOA: savedSinhVien.KHOA,

        DIEMTB: savedSinhVien.DIEMTB,

        LOP: savedSinhVien.LOP,

        EMAIL: savedSinhVien.EMAIL,

        VAITRO: savedSinhVien.VAITRO,
      },
    };
  }

  // =========================
  // ĐĂNG NHẬP
  // =========================

  async login(data: LoginDto) {
    const email = data.EMAIL.trim();

    // =========================
    // TÌM SINH VIÊN
    // =========================

    const sinhVien =
      await this.sinhVienRepository
        .createQueryBuilder('sinhVien')

        .addSelect(
          'sinhVien.MATKHAU',
        )

        .where(
          'LOWER(sinhVien.EMAIL) = LOWER(:email)',
          {
            email,
          },
        )

        .getOne();

    // =========================
    // KHÔNG TÌM THẤY
    // =========================

    if (
      !sinhVien ||
      !sinhVien.MATKHAU
    ) {
      throw new UnauthorizedException(
        'Email hoặc mật khẩu không đúng',
      );
    }

    // =========================
    // KIỂM TRA MẬT KHẨU
    // =========================

    const passwordValid =
      await bcrypt.compare(
        data.MATKHAU,
        sinhVien.MATKHAU,
      );

    if (!passwordValid) {
      throw new UnauthorizedException(
        'Email hoặc mật khẩu không đúng',
      );
    }

    // =========================
    // TẠO JWT PAYLOAD
    // =========================

    const payload = {
      sub: sinhVien.MSV,

      email: sinhVien.EMAIL,

      role: sinhVien.VAITRO,
    };

    // =========================
    // TẠO ACCESS TOKEN
    // =========================

    const accessToken =
      await this.jwtService.signAsync(
        payload,
      );

    // =========================
    // TRẢ KẾT QUẢ
    // =========================

    return {
      message:
        'Đăng nhập thành công',

      access_token: accessToken,

      user: {
        MSV: sinhVien.MSV,

        TENSV: sinhVien.TENSV,

        KHOA: sinhVien.KHOA,

        DIEMTB: sinhVien.DIEMTB,

        LOP: sinhVien.LOP,

        EMAIL: sinhVien.EMAIL,

        VAITRO: sinhVien.VAITRO,
      },
    };
  }
}

import {
  BadRequestException,
  ConflictException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { HttpService } from '@nestjs/axios';
import { InjectRepository } from '@nestjs/typeorm';
import { firstValueFrom } from 'rxjs';
import { Repository } from 'typeorm';

import { DangKy } from './entities/dangky.entity.js';
import { CreateDangKyDto } from './dto/create-dangky.dto.js';
import { UpdateDangKyDto } from './dto/update-dangky.dto.js';

@Injectable()
export class DangkyService {
  private readonly sinhVienUrl =
    process.env.SINHVIEN_SERVICE_URL ?? 'http://localhost:3001';

  private readonly deTaiUrl =
    process.env.DETAI_SERVICE_URL ?? 'http://localhost:3002';

  constructor(
    @InjectRepository(DangKy)
    private readonly dangKyRepository: Repository<DangKy>,

    private readonly httpService: HttpService,
  ) {}

  /**
   * Lấy tất cả đăng ký
   */
  async findAll(): Promise<DangKy[]> {
    return this.dangKyRepository.find();
  }

  /**
   * Lấy một đăng ký theo đề tài + sinh viên
   */
  async findOne(
    detaiId: number,
    msv: string,
  ): Promise<DangKy> {
    const dangKy = await this.dangKyRepository.findOne({
      where: {
        DETAI_ID: detaiId,
        MSV: msv,
      },
    });

    if (!dangKy) {
      throw new NotFoundException(
        `Không tìm thấy đăng ký của sinh viên ${msv} với đề tài ${detaiId}`,
      );
    }

    return dangKy;
  }

  /**
   * Lấy danh sách đề tài mà sinh viên đã đăng ký
   */
  async findBySinhVien(msv: string): Promise<DangKy[]> {
    return this.dangKyRepository.find({
      where: {
        MSV: msv,
      },
    });
  }

  /**
   * Lấy danh sách sinh viên đăng ký một đề tài
   */
  async findByDetai(detaiId: number): Promise<DangKy[]> {
    return this.dangKyRepository.find({
      where: {
        DETAI_ID: detaiId,
      },
    });
  }

  /**
   * Gọi SINHVIEN Service để kiểm tra sinh viên
   */
  private async validateSinhVien(msv: string) {
    try {
      const response = await firstValueFrom(
        this.httpService.get(
          `${this.sinhVienUrl}/sinhvien/${encodeURIComponent(msv)}`,
      ));

      return response.data;
    } catch {
      throw new BadRequestException(
        `Không thể xác thực sinh viên ${msv}`,
      );
    }
  }

  /**
   * Gọi DETAI Service để kiểm tra đề tài

      );

} from '@nestjs/common';

import { InjectRepository } from '@nestjs/typeorm';
import { Repository, Like } from 'typeorm';

import { Detai } from './entities/detai.entity.js';
import { CreateDetaiDto, UpdateDetaiDto } from './dto/detai.dto.js';

@Injectable()
export class DetaiService {
  constructor(
    @InjectRepository(Detai)
    private readonly detaiRepo: Repository<Detai>,
  ) {}

  // Thêm đề tài
  async create(dto: CreateDetaiDto): Promise<Detai> {
    const existingDetai = await this.detaiRepo.findOne({
      where: {
        tenDetai: dto.tenDetai,
      },
    });
    if (existingDetai) {
      throw new ConflictException(`Đề tài "${dto.tenDetai}" đã tồn tại`);
    }
    const detai = this.detaiRepo.create({
      ...dto,
      trangThai: 0,
    });

    return await this.detaiRepo.save(detai);
  }

  // Lấy danh sách đề tài
  async findAll(): Promise<Detai[]> {
    return await this.detaiRepo.find();
  }

  // Tìm kiếm đề tài theo từ khóa
  async search(keyword: string): Promise<Detai[]> {
    return await this.detaiRepo.find({
      where: [
        { tenDetai: Like(`%${keyword}%`) },
        { moTa: Like(`%${keyword}%`) },
      ],
    });
  }

  // Lấy chi tiết đề tài theo ID
  async findOne(MDT: number): Promise<Detai> {
    const detai = await this.detaiRepo.findOneBy({ MDT });

    if (!detai) {
      throw new NotFoundException(`Không tìm thấy đề tài với id = ${MDT}`);
    }

    return detai;
  }

  // Cập nhật thông tin đề tài
  async update(id: number, dto: UpdateDetaiDto): Promise<Detai> {
    const detai = await this.findOne(id);

    if (
      dto.trangThai !== undefined &&
      dto.trangThai !== 0 &&
      dto.trangThai !== 1
    ) {
      throw new BadRequestException('Trạng thái chỉ được phép là 0 hoặc 1');
    }

    Object.assign(detai, dto);

    return await this.detaiRepo.save(detai);
  }

  // Thay đổi trạng thái đề tài
  async updateStatus(id: number, trangThai: number): Promise<Detai> {
    const detai = await this.findOne(id);

    if (trangThai !== 0 && trangThai !== 1) {
      throw new BadRequestException('Trạng thái chỉ được phép là 0 hoặc 1');
    }

    detai.trangThai = trangThai;

    return await this.detaiRepo.save(detai);
  }

  // Xóa đề tài
  async remove(id: number): Promise<{ deleted: boolean }> {
    const result = await this.detaiRepo.delete(id);

    if (result.affected === 0) {
      throw new NotFoundException(`Không tìm thấy đề tài với id = ${id}`);
    }

    return { deleted: true };
  }
}

import {
  CanActivate,
  ExecutionContext,
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';

import { JwtService } from '@nestjs/jwt';
import type { Request } from 'express';

@Injectable()
export class JwtAuthGuard implements CanActivate {
  constructor(
    private readonly jwtService: JwtService,
  ) {}

  async canActivate(
    context: ExecutionContext,
  ): Promise<boolean> {
    const request =
      context.switchToHttp().getRequest<Request>();

    const authHeader =
      request.headers.authorization;

    if (!authHeader) {
      throw new UnauthorizedException(
        'Chưa đăng nhập',
      );
    }

    const [type, token] =
      authHeader.split(' ');

    if (
      type !== 'Bearer' ||
      !token
    ) {
      throw new UnauthorizedException(
        'Token không hợp lệ',
      );
    }

    try {
      const payload =
        await this.jwtService.verifyAsync(token);

      (
        request as Request & {
          user: any;
        }
      ).user = payload;

      return true;
    } catch {
      throw new UnauthorizedException(
        'Token không hợp lệ hoặc đã hết hạn',
      );
    }
  }
}

import {
  CanActivate,
  ExecutionContext,
  ForbiddenException,
  Injectable,
} from '@nestjs/common';

import { Reflector } from '@nestjs/core';

import { ROLES_KEY } from '../decorators/roles.decorator.js';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(
    private readonly reflector: Reflector,
  ) {}

  canActivate(
    context: ExecutionContext,
  ): boolean {
    const requiredRoles =
      this.reflector.getAllAndOverride<string[]>(
        ROLES_KEY,
        [
          context.getHandler(),
          context.getClass(),
        ],
      );

    if (!requiredRoles) {
      return true;
    }

    const request =
      context.switchToHttp().getRequest();

    const user = request.user;

    if (!user) {
      throw new ForbiddenException(
        'Không tìm thấy thông tin người dùng',
      );
    }

    if (!requiredRoles.includes(user.role)) {
      throw new ForbiddenException(
        'Bạn không có quyền truy cập chức năng này',
      );
    }

    return true;
  }
}

import {
  Injectable,
  NotFoundException,
} from '@nestjs/common';

import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';

import { SinhVien } from './entities/sinhvien.entity.js';

@Injectable()
export class SinhvienService {
  constructor(
    @InjectRepository(SinhVien)
    private readonly sinhVienRepository: Repository<SinhVien>,
  ) {}

  // GET /sinhvien
  async findAll(): Promise<SinhVien[]> {
    return this.sinhVienRepository.find();
  }

  // GET /sinhvien/:msv
  async findOne(msv: string): Promise<SinhVien> {
    const sinhVien = await this.sinhVienRepository.findOne({
      where: {
        MSV: msv,
      },
    });

    if (!sinhVien) {
      throw new NotFoundException(
        `Không tìm thấy sinh viên có mã ${msv}`,
      );
    }

    return sinhVien;
  }

  // POST /sinhvien
  async create(data: Partial<SinhVien>): Promise<SinhVien> {
    const sinhVien = this.sinhVienRepository.create(data);

    return this.sinhVienRepository.save(sinhVien);
  }

  // PUT /sinhvien/:msv
  async update(
    msv: string,
    data: Partial<SinhVien>,
  ): Promise<SinhVien> {
    const sinhVien = await this.findOne(msv);

    Object.assign(sinhVien, data);

    return this.sinhVienRepository.save(sinhVien);
  }

  // DELETE /sinhvien/:msv
  async remove(msv: string): Promise<{ message: string }> {
    const sinhVien = await this.findOne(msv);

    await this.sinhVienRepository.remove(sinhVien);

    return {
      message: `Đã xóa sinh viên có mã ${msv}`,
    };
  }
}
